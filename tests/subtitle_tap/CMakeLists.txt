add_executable(subtitle_tap_tests tap_tests.cpp)
find_package(Threads REQUIRED)
target_link_libraries(subtitle_tap_tests PRIVATE subtitle_tap Threads::Threads)
target_compile_definitions(subtitle_tap_tests PRIVATE SUBREC_TAP_HAS_FFMPEG=$<BOOL:${SUBREC_TAP_HAS_FFMPEG}>)
set(tap_cases lifecycle)
if(SUBREC_TAP_HAS_FFMPEG)
    target_include_directories(subtitle_tap_tests PRIVATE "${SUBREC_TAP_FFMPEG_INCLUDE}")
    list(APPEND tap_cases formats overflow_wrap splice_directions max_push_ns spsc_million spsc_million_drops)
else()
    list(APPEND tap_cases fallback_rejects)
endif()
foreach(case IN LISTS tap_cases)
    add_test(NAME subtitle_tap_${case} COMMAND subtitle_tap_tests ${case})
    set_tests_properties(subtitle_tap_${case} PROPERTIES TIMEOUT 60 LABELS "subtitle;tap")
endforeach()
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/SubRecTests.cmake")
subrec_configure_tests()
