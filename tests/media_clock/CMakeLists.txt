cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(media_clock_contract_tests LANGUAGES CXX)
    enable_testing()
endif()
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/SubRecFFmpeg.cmake")
subrec_require_ffmpeg_root()
add_executable(media_clock_contract_compile contract_compile.cpp)
target_compile_features(media_clock_contract_compile PRIVATE cxx_std_17)
target_include_directories(media_clock_contract_compile PRIVATE ../.. "${FFMPEG_ROOT}/include")
if(WIN32)
    target_link_libraries(media_clock_contract_compile PRIVATE "${FFMPEG_ROOT}/lib/avutil.lib")
else()
    find_library(contract_avutil NAMES avutil PATHS "${FFMPEG_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
    target_link_libraries(media_clock_contract_compile PRIVATE "${contract_avutil}")
endif()
add_test(NAME media_clock_contract COMMAND media_clock_contract_compile)
set_tests_properties(media_clock_contract PROPERTIES TIMEOUT 10 LABELS "clock;contract")
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/SubRecTests.cmake")
subrec_configure_tests()
