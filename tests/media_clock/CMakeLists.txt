cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(media_clock_contract_tests LANGUAGES CXX)
    enable_testing()
endif()
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/SubRecFFmpeg.cmake")
subrec_require_ffmpeg_root()
add_executable(media_clock_contract_compile contract_compile.cpp)
target_compile_features(media_clock_contract_compile PRIVATE cxx_std_17)
target_include_directories(media_clock_contract_compile PRIVATE ../.. "${FFMPEG_ROOT}/include")
if(WIN32)
    set(contract_avutil "${FFMPEG_ROOT}/lib/avutil.lib")
else()
    find_library(contract_avutil NAMES avutil PATHS "${FFMPEG_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
endif()
target_link_libraries(media_clock_contract_compile PRIVATE "${contract_avutil}")
add_test(NAME media_clock_contract COMMAND media_clock_contract_compile)
set_tests_properties(media_clock_contract PROPERTIES TIMEOUT 10 LABELS "clock;contract")
find_package(Threads REQUIRED)
add_executable(media_clock_tests media_clock_tests.cpp ../../queue/media_clock.cpp ../../queue/video_tick_scheduler.cpp)
target_compile_features(media_clock_tests PRIVATE cxx_std_17)
target_include_directories(media_clock_tests PRIVATE ../.. "${FFMPEG_ROOT}/include")
target_link_libraries(media_clock_tests PRIVATE Threads::Threads "${contract_avutil}")
foreach(mode IN ITEMS clock 30 ntsc segments collisions)
    add_test(NAME media_clock_${mode} COMMAND media_clock_tests ${mode})
    set_tests_properties(media_clock_${mode} PROPERTIES TIMEOUT 30 LABELS "clock;sync")
endforeach()
add_executable(pc3_regression_tests pc3_regression_tests.cpp ../../queue/media_clock.cpp ../../queue/video_tick_scheduler.cpp)
target_compile_features(pc3_regression_tests PRIVATE cxx_std_17)
target_include_directories(pc3_regression_tests PRIVATE ../.. "${FFMPEG_ROOT}/include")
target_link_libraries(pc3_regression_tests PRIVATE Threads::Threads "${contract_avutil}")
foreach(mode IN ITEMS jitter pauses stamps)
    add_test(NAME media_clock_pc3_${mode} COMMAND pc3_regression_tests ${mode})
    set_tests_properties(media_clock_pc3_${mode} PROPERTIES TIMEOUT 30 LABELS "ci;clock;regression")
endforeach()
add_executable(qpc_present_tests qpc_present_tests.cpp ../../capture/qpc_session_clock.cpp
    ../../queue/media_clock.cpp ../../queue/video_tick_scheduler.cpp)
target_compile_features(qpc_present_tests PRIVATE cxx_std_17)
target_include_directories(qpc_present_tests PRIVATE ../.. "${FFMPEG_ROOT}/include")
target_link_libraries(qpc_present_tests PRIVATE Threads::Threads "${contract_avutil}")
foreach(mode IN ITEMS qpc present)
    add_test(NAME media_clock_${mode} COMMAND qpc_present_tests ${mode})
    set_tests_properties(media_clock_${mode} PROPERTIES TIMEOUT 30 LABELS "ci;clock;present")
endforeach()
find_package(Qt6 REQUIRED COMPONENTS Core)
add_executable(audio_chunk_tests audio_chunk_tests.cpp ../../queue/recordersyncqueue.cpp)
target_compile_features(audio_chunk_tests PRIVATE cxx_std_17)
target_include_directories(audio_chunk_tests PRIVATE ../.. "${FFMPEG_ROOT}/include")
target_link_libraries(audio_chunk_tests PRIVATE Qt6::Core "${contract_avutil}")
add_test(NAME media_clock_audio_chunks COMMAND audio_chunk_tests)
set_tests_properties(media_clock_audio_chunks PROPERTIES LABELS "audio;sync" TIMEOUT 10)

add_executable(media_clock_opaque_props_tests opaque_props_tests.cpp)
target_compile_features(media_clock_opaque_props_tests PRIVATE cxx_std_17)
target_include_directories(media_clock_opaque_props_tests PRIVATE ../.. "${FFMPEG_ROOT}/include")
if(WIN32)
    target_link_libraries(media_clock_opaque_props_tests PRIVATE
        "${FFMPEG_ROOT}/lib/avfilter.lib" "${FFMPEG_ROOT}/lib/avutil.lib")
else()
    find_library(opaque_avfilter NAMES avfilter PATHS "${FFMPEG_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
    target_link_libraries(media_clock_opaque_props_tests PRIVATE "${opaque_avfilter}" "${contract_avutil}")
endif()
add_test(NAME media_clock_opaque_props COMMAND media_clock_opaque_props_tests)
set_tests_properties(media_clock_opaque_props PROPERTIES TIMEOUT 30 LABELS "ci;clock;metadata" SKIP_RETURN_CODE 77)

include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/SubRecTests.cmake")
subrec_configure_tests()
