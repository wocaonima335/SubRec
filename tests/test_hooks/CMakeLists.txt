cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(test_hook_tests LANGUAGES CXX)
    enable_testing()
endif()
foreach(mode IN ITEMS production enabled)
    add_executable(test_hook_${mode}_tests test_hook_tests.cpp)
    target_compile_features(test_hook_${mode}_tests PRIVATE cxx_std_17)
    target_include_directories(test_hook_${mode}_tests PRIVATE ../..)
    target_compile_definitions(test_hook_${mode}_tests PRIVATE slots= signals=public emit=)
    if(mode STREQUAL "enabled")
        target_compile_definitions(test_hook_${mode}_tests PRIVATE SUBREC_ENABLE_TEST_HOOKS=1)
    endif()
    add_test(NAME test_hooks_${mode} COMMAND test_hook_${mode}_tests)
    set_tests_properties(test_hooks_${mode} PROPERTIES TIMEOUT 10)
endforeach()
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/SubRecTests.cmake")
subrec_configure_tests()
