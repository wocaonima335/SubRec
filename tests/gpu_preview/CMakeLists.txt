cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(gpu_preview_tests LANGUAGES CXX)
    enable_testing()
endif()
find_package(Threads REQUIRED)
add_executable(preview_tests preview_tests.cpp)
target_compile_features(preview_tests PRIVATE cxx_std_17)
target_include_directories(preview_tests PRIVATE ../..)
target_link_libraries(preview_tests PRIVATE Threads::Threads)
add_test(NAME preview_latest_slot COMMAND preview_tests)
set_tests_properties(preview_latest_slot PROPERTIES TIMEOUT 10)
if(FFMPEG_ROOT)
    add_executable(preview_graph_tests preview_graph_tests.cpp)
    target_compile_features(preview_graph_tests PRIVATE cxx_std_17)
    target_include_directories(preview_graph_tests PRIVATE ../.. "${FFMPEG_ROOT}/include")
    foreach(component avfilter avutil)
        find_library(PREVIEW_${component} NAMES ${component} PATHS "${FFMPEG_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
        target_link_libraries(preview_graph_tests PRIVATE "${PREVIEW_${component}}")
    endforeach()
    add_test(NAME preview_graph COMMAND preview_graph_tests)
    set_tests_properties(preview_graph PROPERTIES TIMEOUT 30)
endif()
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/SubRecTests.cmake")
subrec_configure_tests()
