cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(startup_failure_tests LANGUAGES CXX)
    enable_testing()
    find_package(Qt6 REQUIRED COMPONENTS Core)
    include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/SubRecFFmpeg.cmake")
    subrec_require_ffmpeg_root()
endif()
find_package(Threads REQUIRED)
add_executable(startup_failure_tests startup_failure_tests.cpp
    ../../recorder/capture_session.cpp ../../thread/ffscreengrabthread.cpp
    ../../thread/ffthread.cpp ../../queue/ffvframequeue.cpp)
target_compile_features(startup_failure_tests PRIVATE cxx_std_17)
target_include_directories(startup_failure_tests PRIVATE fakes ../.. "${FFMPEG_ROOT}/include")
target_link_libraries(startup_failure_tests PRIVATE Qt6::Core Threads::Threads
    "${FFMPEG_ROOT}/lib/avutil.lib")
# Exercise the same keyword macros as QtCore on PC3, including standalone builds.
target_compile_definitions(startup_failure_tests PRIVATE slots= signals=public emit=)
add_test(NAME video_startup_failure COMMAND startup_failure_tests)
add_test(NAME screen_normal_drain COMMAND startup_failure_tests --normal-drain)
set_tests_properties(video_startup_failure screen_normal_drain PROPERTIES TIMEOUT 10)
if(WIN32)
    get_filename_component(_startup_compiler_bin "${CMAKE_CXX_COMPILER}" DIRECTORY)
    set(_startup_path "${FFMPEG_ROOT}/bin;${_startup_compiler_bin};$ENV{PATH}")
    get_target_property(_startup_qt_type Qt6::Core TYPE)
    if(NOT _startup_qt_type STREQUAL "INTERFACE_LIBRARY")
        set(_startup_path "$<TARGET_FILE_DIR:Qt6::Core>;${_startup_path}")
    endif()
    string(REPLACE ";" "\\;" _startup_path "${_startup_path}")
    set_tests_properties(video_startup_failure screen_normal_drain PROPERTIES ENVIRONMENT "PATH=${_startup_path}")
endif()
foreach(mode IN ITEMS stuck-audio stuck-video autotest-exit)
    if(NOT mode STREQUAL "autotest-exit")
        set(expected_code 6)
    else()
        set(expected_code 3)
    endif()
    add_test(NAME startup_${mode} COMMAND "${CMAKE_COMMAND}"
        "-DTEST_EXE=$<TARGET_FILE:startup_failure_tests>" "-DTEST_MODE=--${mode}"
        "-DEXPECTED_CODE=${expected_code}" -P "${CMAKE_CURRENT_LIST_DIR}/exit_deadline_test.cmake")
    set_tests_properties(startup_${mode} PROPERTIES TIMEOUT 5)
    if(WIN32)
        set_tests_properties(startup_${mode} PROPERTIES ENVIRONMENT "PATH=${_startup_path}")
    endif()
endforeach()

include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/SubRecTests.cmake")
subrec_configure_tests()
