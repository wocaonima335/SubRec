cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(subrec_logging_tests LANGUAGES CXX)
    enable_testing()
    find_package(Qt6 QUIET COMPONENTS Core)
endif()
add_executable(log_options_tests log_options_tests.cpp)
target_compile_features(log_options_tests PRIVATE cxx_std_17)
target_include_directories(log_options_tests PRIVATE ../..)
set(log_test_targets log_options_tests)
if(TARGET Qt6::Core)
    add_executable(qt_frame_log_tests qt_frame_log_tests.cpp)
    target_compile_features(qt_frame_log_tests PRIVATE cxx_std_17)
    target_include_directories(qt_frame_log_tests PRIVATE ../..)
    target_link_libraries(qt_frame_log_tests PRIVATE Qt6::Core)
    list(APPEND log_test_targets qt_frame_log_tests)
else()
    message(STATUS "Qt6 Core SDK unavailable: building option tests only; Qt emission tests require the application Qt kit")
endif()
foreach(target IN LISTS log_test_targets)
    target_compile_definitions(${target} PRIVATE slots= signals=public emit=)
    foreach(mode IN ITEMS default env-debug env-level cli invalid cli-overrides-env)
        set(test_args)
        set(debug_env "")
        set(level_env "")
        # Even an external Qt rule must not accidentally enable verbose capture.
        if(mode STREQUAL "env-debug")
            set(debug_env "1")
            list(APPEND test_args --expect-debug)
        elseif(mode STREQUAL "env-level")
            set(level_env "debug")
            list(APPEND test_args --expect-debug)
        elseif(mode STREQUAL "cli")
            list(APPEND test_args --debug-log --expect-debug)
        elseif(mode STREQUAL "invalid")
            set(debug_env "invalid")
            set(level_env "invalid")
        elseif(mode STREQUAL "cli-overrides-env")
            set(debug_env "0")
            set(level_env "info")
            list(APPEND test_args --debug-log --expect-debug)
        endif()
        set(test_env "SUBREC_DEBUG_LOG=${debug_env}" "SUBREC_LOG_LEVEL=${level_env}"
            "QT_LOGGING_RULES=subrec.frames.debug=true" "QT_FATAL_WARNINGS=" "QT_FATAL_CRITICALS=")
        add_test(NAME ${target}_${mode} COMMAND "${CMAKE_COMMAND}" -E env
            ${test_env} "$<TARGET_FILE:${target}>" ${test_args})
        set_tests_properties(${target}_${mode} PROPERTIES TIMEOUT 10)
        if(WIN32 AND TARGET Qt6::Core)
            get_target_property(qt_type Qt6::Core TYPE)
            if(NOT qt_type STREQUAL "INTERFACE_LIBRARY")
                get_filename_component(compiler_bin "${CMAKE_CXX_COMPILER}" DIRECTORY)
                set(runtime_path "$<TARGET_FILE_DIR:Qt6::Core>;${compiler_bin};$ENV{PATH}")
                string(REPLACE ";" "\\;" runtime_path "${runtime_path}")
                set_tests_properties(${target}_${mode} PROPERTIES ENVIRONMENT "PATH=${runtime_path}")
            endif()
        endif()
    endforeach()
endforeach()

include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/SubRecTests.cmake")
subrec_configure_tests()
