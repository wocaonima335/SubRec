cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(encoder_tuning_tests LANGUAGES CXX)
    enable_testing()
endif()
add_executable(encoder_tuning_tests encoder_tuning_tests.cpp)
target_compile_features(encoder_tuning_tests PRIVATE cxx_std_17)
target_include_directories(encoder_tuning_tests PRIVATE ../..)
add_test(NAME encoder_tuning COMMAND encoder_tuning_tests)
set_tests_properties(encoder_tuning PROPERTIES TIMEOUT 10 LABELS "encoder")
if(FFMPEG_ROOT)
    add_executable(encoder_options_tests encoder_options_tests.cpp)
    target_compile_features(encoder_options_tests PRIVATE cxx_std_17)
    target_include_directories(encoder_options_tests PRIVATE ../.. "${FFMPEG_ROOT}/include")
    if(WIN32)
        target_link_libraries(encoder_options_tests PRIVATE
            "${FFMPEG_ROOT}/lib/avcodec.lib" "${FFMPEG_ROOT}/lib/avutil.lib")
    else()
        foreach(component avcodec avutil)
            find_library(ENCODER_TUNING_${component} NAMES ${component} PATHS "${FFMPEG_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
            target_link_libraries(encoder_options_tests PRIVATE "${ENCODER_TUNING_${component}}")
        endforeach()
    endif()
    add_test(NAME encoder_options COMMAND encoder_options_tests)
    # 77: none of the H.264 encoders is in this FFmpeg build.
    set_tests_properties(encoder_options PROPERTIES TIMEOUT 30 LABELS "encoder" SKIP_RETURN_CODE 77)
endif()
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/SubRecTests.cmake")
subrec_configure_tests()
